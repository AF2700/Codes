import heapq
def huffman(chars,freq):
    heap=[]
    for i in range(len(chars)):
        heap.append ((freq[i],chars[i]))
    heapq.heapify(heap)
    while len(heap)>1:
        a=heapq.heappop(heap)
        b=heapq.heappop(heap)
        heapq.heappush(heap,(a[0]+b[0],[a[1],b[1]]))
    def show(tree,code=""):
        if isinstance(tree,str):
            print(tree, ":", code)
        else:
            show(tree[0],code + "0")
            show(tree[1],code + "1")
    show(heap[0][1])

n = int(input("Enter number of items: "))

w = list(map(int, input("Enter weights: ").split()))
p = list(map(int, input("Enter profits: ").split()))

capacity = int(input("Enter capacity: "))

for i in range(n):
    for j in range(i+1, n):
        if p[i]/w[i] < p[j]/w[j]:
            w[i], w[j] = w[j], w[i]
            p[i], p[j] = p[j], p[i]

total = 0

for i in range(n):
    if capacity >= w[i]:
        capacity -= w[i]
        total += p[i]
        print("Item", i+1, ":", w[i], "taken")
    else:
        total += (p[i]/w[i]) * capacity
        print("Item", i+1, ":", capacity, "taken")
        break

print("Maximum profit =", total)

weight = [10, 20, 30]
profit = [60, 100, 120]
capacity = 50
n = len(weight)

dp = [[0]*(capacity+1) for i in range(n+1)]

for i in range(1, n+1):
    for w in range(1, capacity+1):
        if weight[i-1] <= w:
            dp[i][w] = max(profit[i-1] + dp[i-1][w-weight[i-1]],
                           dp[i-1][w])
        else:
            dp[i][w] = dp[i-1][w]

# Display DP Grid
print("Dynamic Programming Grid:")
for row in dp:
    print(row)

# Display Final Profit
print("\nMaximum Profit =", dp[n][capacity])
